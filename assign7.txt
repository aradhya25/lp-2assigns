pragma solidity ^0.8.20;
contract studentrecord{
    uint public rollno;
    string public name;
    uint public marks;
    function setstudent(uint r,string memory n,uint m) public{
        rollno=r;
        name=n;
        marks=m;
    }
    function getstudent() public view returns(uint,string memory,uint){
        return(rollno,name,marks);
    }
}
